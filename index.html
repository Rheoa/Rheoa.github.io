async function rewriteDynamicLinks(articleId) { const country = await getCountryCode(); const { data: links, error } = await supabaseClient .from('affiliate_links') .select('affiliate_key, link_url, country_code') .eq('article_id', articleId) .or(`country_code.eq.${country},country_code.eq.default`); if (error || !links) return; const linkMap = {}; links.forEach(item => { const existing = linkMap[item.affiliate_key]; if (!existing || (item.country_code !== 'default' && existing === 'default')) { linkMap[item.affiliate_key] = item.link_url; } }); document.querySelectorAll('.dynamic-link').forEach(el => { const key = el.dataset.affiliateKey; if (key && linkMap[key]) { el.href = linkMap[key]; } else { el.style.opacity = '0.5'; el.style.cursor = 'not-allowed'; } }); }
